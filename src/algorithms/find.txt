   #include <algorithm>
   #include <concepts>
   #include <iostream>
   #include <iterator>
   #include <numeric>
   #include <vector>

   // Find the first value in [first, last) equal to value.
   template <std::input_iterator InputIt, class T>
   requires std::equality_comparable_with<
       std::iter_reference_t<InputIt>, const T&>
   InputIt my_find(InputIt first, InputIt last, const T& value)
   {
     for (; first != last; ++first) {
       if (*first == value) {
         return first;
       }
     }
     return last;
   }

   int main () {
     int value = 144;
     std::vector<int> nums(999);
     std::iota(nums.begin(), nums.end(), -72);

     auto custom_it = my_find(nums.begin(), nums.end(), value);
     auto standard_it = std::find(nums.begin(), nums.end(), value);

     if (custom_it == standard_it && custom_it != nums.end()) {
       std::cout << "Both algorithms found " << value << " at position "
                 << (std::distance(nums.begin(), custom_it)) << '\n';
     }
     return 0;
   }
