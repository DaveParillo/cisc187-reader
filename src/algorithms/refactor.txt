   #include <concepts>
   #include <iostream>
   #include <iterator>
   #include <list>

   // accumulate the sum of values in range [first, last)
   template <std::input_iterator InputIt, class T>
   requires std::movable<T>
         && requires(T value, std::iter_reference_t<InputIt> element) {
              { value + element } -> std::convertible_to<T>;
            }
   T sum(InputIt first, InputIt last, T value) {
     while (first != last) {
       value = value + *first;
       ++first;
     }
     return value;
   }

   int main() {
     float values[] = {1, 1, 2, 3, 5, 8, 13, 21, 34};
     float* end = values + sizeof(values) / sizeof(*values);
     double total = sum(values, end, 0.0);
     std::cout << "array sum = " << total << '\n';

     std::list<float> linked_values = {1, 1, 2, 3, 5, 8, 13, 21, 34};
     total = sum(linked_values.begin(), linked_values.end(), 0.0);
     std::cout << "list sum = " << total << '\n';
   }
